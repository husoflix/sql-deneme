null değerini kontrol etmek için is null veya is not null ifadeleri kullanılır 

null değerini kontrol etmek için ;
select * from tablo_adı where sütun_adı is null;

null değerini hariç tutma null olmayan değerleri listeleme;

null değerini güncellemek için ;

uptade tablo_adı
set sütun_adı = 'yeni değer'
where sütun_adı is null;
